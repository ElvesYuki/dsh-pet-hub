cmake_minimum_required(VERSION 3.22)
get_filename_component(pet_firmware "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${pet_firmware}/../media/profiles/400x400/playback.cmake")
set(PET_VP8_PROBE ON)
option(VP8_PIPELINE "Use Muse SIMD conversion and a separate panel task" OFF)
option(VP8_RATE_PROBE "Find a sustainable 400px playback rate after the 24fps probe" OFF)
option(VP8_VISUAL_PROBE "Test 400px with conversion on the display core" OFF)
option(VP8_FLASH_PROBE "Download VP8 to the guarded scratch range and replay after reboot" OFF)
option(VP8_RETAINED "Play the four persisted daily actions with touch and automatic transitions" OFF)
option(VP8_FAST_TOUCH "Keep verified idle and wave in RAM and cancel old queued frames" OFF)
option(VP8_ACTION_UPDATE "Enable the guarded one-object VP8 action 1 updater" OFF)
option(VP8_POWER_CUT_PROBE "Use a second update region for a controlled USB power cycle" OFF)
option(VP8_OTA "Build the 1.75C dual-slot OTA and VP8 object cache candidate" OFF)
option(VP8_BOOT_PAIRING "Include the experimental BOOT enrollment and persistent proof" OFF)
option(VP8_USB_ENROLL "Enable explicit first enrollment from a trusted direct USB connection" OFF)
option(VP8_BATTERY_QUERY "Include battery samples in the device status snapshot" OFF)
option(VP8_SENSOR_QUERY "Include motion and touch in the device status snapshot" OFF)
option(VP8_LOCAL_MOTION "Enable local shake and tilt animations" OFF)
option(VP8_SLEEP_WAKE "Enable local sleep animation, dimming and touch wake" OFF)
option(VP8_LIFT_WAKE "Wake the dimmed desktop on a deliberate lift detected by the IMU" OFF)
option(VP8_FOCUS_MENU "Enable PWR launcher and offline focus timer" OFF)
option(VP8_DESKTOP_INTERNET "Enable router-connected desktop SNTP and HTTPS weather" OFF)
option(VP8_VPC1_USB_PROBE "Enable the direct USB VPC1 playback fixture" OFF)
option(VP8_VPC1_LAN "Enable the explicitly armed VP8 LAN control and status candidate" OFF)
option(VP8_TOUCH_SWIPE "Enable local animation browsing with horizontal swipes" OFF)
option(VP8_PLAYFUL_INTERACTIONS "Enable upward-swipe surprises and battery reactions" OFF)
option(VP8_OTA_REJECT_BOOT_TEST "Reject the first OTA boot in an isolated test image" OFF)
option(VP8_OTA_HOLD_CONFIRM_TEST "Hold the first OTA boot before accepting it for a controlled power test" OFF)
option(VP9_REFERENCE_PROBE "Use the upstream generic-C VP9 decoder with the 400px visual pipeline" OFF)
if(VP8_FLASH_PROBE AND (NOT VP8_VISUAL_PROBE OR VP9_REFERENCE_PROBE))
  message(FATAL_ERROR "VP8_FLASH_PROBE requires the VP8 visual pipeline")
endif()
if(VP8_RETAINED AND NOT VP8_FLASH_PROBE)
  message(FATAL_ERROR "VP8_RETAINED requires VP8_FLASH_PROBE")
endif()
if(VP8_FAST_TOUCH AND NOT VP8_RETAINED)
  message(FATAL_ERROR "VP8_FAST_TOUCH requires VP8_RETAINED")
endif()
if(VP8_ACTION_UPDATE AND (NOT VP8_FAST_TOUCH OR NOT EXISTS "${VP8_UPDATE_INPUT}/action_update_expected.h"))
  message(FATAL_ERROR "VP8_ACTION_UPDATE requires VP8_FAST_TOUCH and the approved update header")
endif()
if(VP8_POWER_CUT_PROBE AND NOT VP8_ACTION_UPDATE)
  message(FATAL_ERROR "VP8_POWER_CUT_PROBE requires VP8_ACTION_UPDATE")
endif()
if(VP8_OTA AND (NOT VP8_FAST_TOUCH OR VP8_ACTION_UPDATE OR NOT VP8_OTA_VERSION))
  message(FATAL_ERROR "VP8_OTA requires VP8_FAST_TOUCH and VP8_OTA_VERSION and excludes the fixed-object updater")
endif()
if(VP8_BOOT_PAIRING AND (NOT VP8_OTA OR NOT VP8_VISUAL_PROBE OR NOT VP8_OTA_VERSION MATCHES "-pair-test$"))
  message(FATAL_ERROR "VP8_BOOT_PAIRING requires the visual OTA pipeline and a separate -pair-test version")
endif()
if(VP8_USB_ENROLL AND (NOT VP8_BOOT_PAIRING OR NOT VP8_OTA_VERSION MATCHES "-usb-pair-test$"))
  message(FATAL_ERROR "VP8_USB_ENROLL requires VP8_BOOT_PAIRING and a separate -usb-pair-test version")
endif()
if(VP8_BATTERY_QUERY AND (NOT VP8_USB_ENROLL OR NOT VP8_OTA_VERSION MATCHES "-battery-usb-pair-test$"))
  message(FATAL_ERROR "VP8_BATTERY_QUERY requires VP8_USB_ENROLL and a separate -battery-usb-pair-test version")
endif()
if(VP8_SENSOR_QUERY AND NOT VP8_BATTERY_QUERY)
  message(FATAL_ERROR "VP8_SENSOR_QUERY requires the USB battery query build")
endif()
if(VP8_LOCAL_MOTION AND (NOT VP8_SENSOR_QUERY OR NOT VP8_PLAYFUL_INTERACTIONS))
  message(FATAL_ERROR "VP8_LOCAL_MOTION requires the existing sensor and playful interaction profile")
endif()
if(VP8_SLEEP_WAKE AND NOT VP8_PLAYFUL_INTERACTIONS)
  message(FATAL_ERROR "VP8_SLEEP_WAKE requires the existing playful interaction profile")
endif()
if(VP8_LIFT_WAKE AND (NOT VP8_SLEEP_WAKE OR NOT VP8_LOCAL_MOTION))
  message(FATAL_ERROR "VP8_LIFT_WAKE requires both sleep and local motion")
endif()
if(VP8_FOCUS_MENU AND NOT VP8_SLEEP_WAKE)
  message(FATAL_ERROR "VP8_FOCUS_MENU requires the daily sleep and wake profile")
endif()
if(VP8_DESKTOP_INTERNET AND NOT VP8_FOCUS_MENU)
  message(FATAL_ERROR "VP8_DESKTOP_INTERNET requires the desktop and focus profile")
endif()
if(VP8_VPC1_USB_PROBE AND (NOT VP8_OTA OR NOT VP8_FOCUS_MENU OR NOT VP8_OTA_VERSION MATCHES "-pair-test$"))
  message(FATAL_ERROR "VP8_VPC1_USB_PROBE requires the OTA focus profile and a test version")
endif()
if(VP8_VPC1_LAN AND (VP8_VPC1_USB_PROBE OR NOT VP8_OTA OR NOT VP8_BOOT_PAIRING OR
   NOT VP8_DESKTOP_INTERNET OR NOT VP8_BATTERY_QUERY OR NOT VP8_SENSOR_QUERY OR NOT VP8_LOCAL_MOTION OR
   NOT VP8_OTA_VERSION MATCHES "-pair-test$"))
  message(FATAL_ERROR "VP8_VPC1_LAN requires the bound desktop sensor test profile and excludes the USB fixture")
endif()
if(VP8_OTA)
  string(LENGTH "${VP8_OTA_VERSION}" status_version_length)
  if(status_version_length GREATER 31 OR NOT VP8_OTA_VERSION MATCHES "^[A-Za-z0-9][A-Za-z0-9._+-]*$")
    message(FATAL_ERROR "VP8_OTA requires a 1–31 byte Status v2 firmware version")
  endif()
endif()
if(VP8_TOUCH_SWIPE AND NOT VP8_OTA)
  message(FATAL_ERROR "VP8_TOUCH_SWIPE requires the OTA animation cache")
endif()
if(VP8_PLAYFUL_INTERACTIONS AND (NOT VP8_TOUCH_SWIPE OR NOT VP8_BATTERY_QUERY))
  message(FATAL_ERROR "VP8_PLAYFUL_INTERACTIONS requires touch swipes and the battery query")
endif()
if(VP8_OTA_REJECT_BOOT_TEST AND (NOT VP8_OTA OR NOT VP8_OTA_VERSION MATCHES "-test$"))
  message(FATAL_ERROR "VP8_OTA_REJECT_BOOT_TEST requires an OTA image with a -test version")
endif()
if(VP8_OTA_HOLD_CONFIRM_TEST AND (NOT VP8_OTA OR VP8_OTA_REJECT_BOOT_TEST OR NOT VP8_OTA_VERSION MATCHES "-power-test$"))
  message(FATAL_ERROR "VP8_OTA_HOLD_CONFIRM_TEST requires a separate OTA image with a -power-test version")
endif()
if(VP9_REFERENCE_PROBE)
  if(NOT VP8_VISUAL_PROBE OR NOT EXISTS "${VP9_BUILD}/libvpx_srcs.txt" OR NOT EXISTS "${VP9_SOURCE}/vpx/vp8dx.h")
    message(FATAL_ERROR "VP9 requires the visual pipeline, generated libvpx build files and the pinned upstream source")
  endif()
  execute_process(COMMAND git -C "${VP9_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE vp9_revision
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT vp9_revision STREQUAL "1024874c5919305883187e2953de8fcb4c3d7fa6")
    message(FATAL_ERROR "VP9 libvpx revision mismatch")
  endif()
  execute_process(COMMAND git -C "${VP9_SOURCE}" diff --quiet HEAD COMMAND_ERROR_IS_FATAL ANY)
endif()
if(VP8_RATE_PROBE AND NOT VP8_PIPELINE)
  message(FATAL_ERROR "VP8_RATE_PROBE requires VP8_PIPELINE")
endif()
if(VP8_VISUAL_PROBE AND (NOT VP8_PIPELINE OR VP8_RATE_PROBE))
  message(FATAL_ERROR "VP8_VISUAL_PROBE requires VP8_PIPELINE and excludes VP8_RATE_PROBE")
endif()
if(NOT VP8_MUSE)
  message(FATAL_ERROR "Pass the pinned Muse checkout as VP8_MUSE")
endif()
if(NOT VP8_OTA AND (NOT VP8_INPUT OR NOT EXISTS "${VP8_INPUT}/expected.h"))
  message(FATAL_ERROR "The playback probe requires VP8_INPUT with generated expected.h")
endif()
execute_process(COMMAND git -C "${VP8_MUSE}" rev-parse HEAD OUTPUT_VARIABLE muse OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT muse STREQUAL "7b302e37561652eefcc8a4d58085de6f7f800f29")
  message(FATAL_ERROR "Muse revision mismatch")
endif()
execute_process(COMMAND git -C "${VP8_MUSE}" diff --quiet HEAD -- esp32/components/libvpx_vp8
    esp32/components/muse/muse_yuv.c esp32/components/muse/muse_yuv.h esp32/components/muse/muse_yuv_pie.S
    COMMAND_ERROR_IS_FATAL ANY)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/board/round_display")
if(VP9_REFERENCE_PROBE)
  list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/diagnostics/vp9/libvpx_vp8")
else()
  list(APPEND EXTRA_COMPONENT_DIRS "${VP8_MUSE}/esp32/components/libvpx_vp8")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "" FORCE)
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.board.defaults"
    "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.cache-line.defaults" "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.player.defaults")
if(VP9_REFERENCE_PROBE)
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/diagnostics/vp9/sdkconfig.defaults")
endif()
if(VP8_FLASH_PROBE)
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.station.defaults"
    "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.cache.defaults")
endif()
if(VP8_OTA)
  set(PROJECT_VER "${VP8_OTA_VERSION}")
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.ota.defaults")
endif()
if(VP8_DESKTOP_INTERNET)
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/config/sdkconfig.weather.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
if(VP8_OTA)
  project(round_pet_175c_ota)
else()
  project(round_vp8_probe)
endif()
